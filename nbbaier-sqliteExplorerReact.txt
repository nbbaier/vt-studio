import { useEffect } from "https://esm.sh/react@18.2.0";
import { generateDefaultQuery } from "../../shared/sqlHelpers.ts";

/**
 * Custom hook for managing table selection and query persistence
 */
export function useTableSelection(
	selectedTable: string | null,
	onCodeChange: (code: string) => void,
) {
	useEffect(() => {
		if (!selectedTable) return;

		// Try to load saved query from localStorage
		const savedQuery = localStorage.getItem(`saved-${selectedTable}`);

		if (savedQuery) {
			onCodeChange(savedQuery);
		} else {
			// Generate default SELECT query
			const defaultQuery = generateDefaultQuery(selectedTable, "table");
			onCodeChange(defaultQuery);
		}
	}, [selectedTable, onCodeChange]);

	// Save query to localStorage when it changes
	const saveQuery = (tableName: string, query: string) => {
		if (tableName && query.trim()) {
			localStorage.setItem(`saved-${tableName}`, query);
		}
	};

	return { saveQuery };
}

import { useState } from "https://esm.sh/react@18.2.0";
import type { QueryResult } from "../../shared/types.ts";

interface UseQueryExecutionResult {
	result: QueryResult | null;
	error: string | null;
	isLoading: boolean;
	executeQuery: (query: string) => Promise<void>;
	clearResults: () => void;
}

/**
 * Custom hook for executing SQL queries
 */
export function useQueryExecution(): UseQueryExecutionResult {
	const [result, setResult] = useState<QueryResult | null>(null);
	const [error, setError] = useState<string | null>(null);
	const [isLoading, setIsLoading] = useState(false);

	const executeQuery = async (query: string) => {
		if (!query.trim()) {
			setError("Query cannot be empty");
			return;
		}

		setIsLoading(true);
		setError(null);
		setResult(null);

		try {
			const response = await fetch("/api/query", {
				method: "POST",
				headers: {
					"Content-Type": "application/json",
				},
				body: JSON.stringify({ query }),
			});

			// Handle authentication errors
			if (response.status === 401) {
				window.location.href = "/auth/login";
				return;
			}

			const data = await response.json();

			if (data.success) {
				setResult(data.data);
			} else {
				setError(data.error || "Query failed");
			}
		} catch (err) {
			setError(err instanceof Error ? err.message : "Unknown error occurred");
		} finally {
			setIsLoading(false);
		}
	};

	const clearResults = () => {
		setResult(null);
		setError(null);
	};

	return {
		result,
		error,
		isLoading,
		executeQuery,
		clearResults,
	};
}

import { useCallback } from "https://esm.sh/react@18.2.0";
import { getErrorMessage } from "../../shared/types.ts";

type DownloadFormat = "csv" | "json";

/**
 * Custom hook for downloading query results
 */
export function useDownload() {
	const downloadQuery = useCallback(
		async (query: string, format: DownloadFormat) => {
			if (!query.trim()) {
				throw new Error("Cannot download: query is empty");
			}

			try {
				const encodedQuery = encodeURIComponent(query);
				const url = `/api/download/${format}?query=${encodedQuery}`;

				// Trigger download by creating a temporary link
				const link = document.createElement("a");
				link.href = url;
				link.download = `results.${format}`;
				document.body.appendChild(link);
				link.click();
				document.body.removeChild(link);
			} catch (err) {
				// Re-throw with a formatted error message
				throw new Error(
					`Failed to download ${format}: ${getErrorMessage(err)}`,
				);
			}
		},
		[],
	);

	return { downloadQuery };
}

import type { ReactCodeMirrorRef } from "https://esm.sh/@uiw/react-codemirror@4.21.21?deps=react@18.2.0";
import { useCallback, useRef, useState } from "https://esm.sh/react@18.2.0";
import { getStatementAtPosition } from "../../shared/sqlHelpers.ts";

interface UseCodeEditorResult {
	code: string;
	setCode: (code: string) => void;
	getActiveStatement: () => string;
	editorRef: React.MutableRefObject<ReactCodeMirrorRef | null>;
}

/**
 * Custom hook for managing code editor state and extracting SQL statements
 */
export function useCodeEditor(): UseCodeEditorResult {
	const [code, setCode] = useState("");
	const editorRef = useRef<ReactCodeMirrorRef | null>(null);

	/**
	 * Get the SQL statement at the current cursor position
	 * Falls back to entire code if cursor position can't be determined
	 */
	const getActiveStatement = useCallback((): string => {
		if (!code.trim()) return "";

		try {
			// Try to get cursor position from CodeMirror editor
			const editor = editorRef.current;
			if (editor?.view?.state) {
				const cursorPosition = editor.view.state.selection.main.head;
				const statement = getStatementAtPosition(code, cursorPosition);
				return statement || code;
			}
		} catch (err) {
			console.error("Error getting cursor position:", err);
		}

		// Fallback to entire code if cursor position unavailable
		return code;
	}, [code]);

	return {
		code,
		setCode,
		getActiveStatement,
		editorRef,
	};
}

// Shared TypeScript types for SQLite Explorer

export type SchemaItemType = "table" | "view";

export interface SchemaItem {
	name: string;
	type: SchemaItemType;
}

// SQL values can be string, number, boolean, null, or Buffer
export type SQLValue = string | number | boolean | null;

export interface QueryResult {
	columns: string[];
	rows: SQLValue[][];
	rowsAffected?: number;
}

export interface QueryResponse {
	success: boolean;
	data?: QueryResult;
	error?: string;
}

export interface SchemaResponse {
	tables: SchemaItem[];
	views: SchemaItem[];
}

export interface ParsedStatement {
	start: number;
	end: number;
	text: string;
	type: string;
}

// Error type guard helper
export function isError(error: unknown): error is Error {
	return error instanceof Error;
}

// Safe error message extractor
export function getErrorMessage(error: unknown): string {
	if (isError(error)) {
		return error.message;
	}
	if (typeof error === "string") {
		return error;
	}
	return "An unknown error occurred";
}

import { useEffect, useState } from "https://esm.sh/react@18.2.0";

export function ThemeToggle() {
	const [isDark, setIsDark] = useState(false);

	// Initialize theme from localStorage or system preference
	useEffect(() => {
		const stored = localStorage.getItem("theme");
		const prefersDark = window.matchMedia(
			"(prefers-color-scheme: dark)",
		).matches;
		const shouldBeDark = stored === "dark" || (!stored && prefersDark);

		setIsDark(shouldBeDark);
		document.documentElement.classList.toggle("dark", shouldBeDark);
	}, []);

	const toggleTheme = () => {
		const newIsDark = !isDark;
		setIsDark(newIsDark);
		document.documentElement.classList.toggle("dark", newIsDark);
		localStorage.setItem("theme", newIsDark ? "dark" : "light");
	};

	return (
		<button
			type="button"
			onClick={toggleTheme}
			className="p-2 rounded-lg bg-sand-200 dark:bg-slate-800 hover:bg-sand-300 dark:hover:bg-slate-700 transition-colors duration-200"
			aria-label={isDark ? "Switch to light mode" : "Switch to dark mode"}
			title={isDark ? "Switch to light mode" : "Switch to dark mode"}
		>
			{isDark ? (
				<svg
					aria-label="Dark mode"
					className="w-5 h-5 text-warm-400"
					fill="none"
					stroke="currentColor"
					viewBox="0 0 24 24"
				>
					<title>Dark mode</title>

/** @jsxImportSource https://esm.sh/react@18.2.0 */

import { useMemo } from "https://esm.sh/react@18.2.0";
import type { SQLValue } from "../../shared/types.ts";

export const PAGE_SIZE = 100;

interface TableProps {
	headers: string[];
	rows: SQLValue[][];
	currentPage: number;
}

export function Table({ headers, rows, currentPage }: TableProps) {
	// Calculate pagination
	const startIndex = (currentPage - 1) * PAGE_SIZE;
	const endIndex = startIndex + PAGE_SIZE;
	const paginatedRows = useMemo(
		() => rows.slice(startIndex, endIndex),
		[rows, startIndex, endIndex],
	);
	const createRows = (rows: SQLValue[][], headers: string[]) => {
		// Check if all rows are empty
		if (rows.every((subArray) => subArray.length === 0)) {
			const emptyRow = headers.map((header, index) => (
				<td
					key={index}
					className="px-4 py-3 text-sm text-sand-500 dark:text-slate-400 italic"
					aria-label={header}
				>
					{header}
				</td>
			));

			return <tr className="bg-sand-50 dark:bg-slate-800/50">{emptyRow}</tr>;
		}

		return rows.map((row, rowIndex) => (
			<tr
				key={rowIndex}
				className="border-b border-sand-200 dark:border-slate-700 hover:bg-sand-50 dark:hover:bg-slate-800/50 transition-colors duration-150"
			>
				{row.map((cell, cellIndex) => (
					<td
						key={cellIndex}
						className="px-4 py-3 text-sm text-slate-700 dark:text-slate-300 font-mono whitespace-nowrap"
					>
						{cell !== null && cell !== undefined ? (
							String(cell)
						) : (
							<span className="text-sand-400 dark:text-slate-500 italic">
								null
							</span>
						)}
					</td>
				))}
			</tr>
		));
	};

	const tableRows = createRows(paginatedRows, headers);

	return (
		<table className="border-collapse w-full">
			<thead>
				<tr>
					{headers.map((header, index) => (
						<th
							key={index}
							className="px-4 py-3 text-left text-xs font-semibold uppercase tracking-wider text-slate-600 dark:text-slate-400 border-b border-sand-300 dark:border-slate-700 whitespace-nowrap sticky top-0 bg-sand-100 dark:bg-slate-800"
						>
							{header}
						</th>
					))}
				</tr>
			</thead>
			<tbody className="bg-white dark:bg-slate-900">{tableRows}</tbody>
		</table>
	);
}

import { useState } from "https://esm.sh/react@18.2.0";
import type { SchemaItem } from "../../shared/types.ts";
import { SearchSVG, TableSVG, ViewSVG } from "./Icons.tsx";

interface TablesListProps {
	tables: SchemaItem[];
	selectedTable: string | null;
	onTableSelect: (name: string) => void;
}

export function TablesList({
	tables,
	selectedTable,
	onTableSelect,
}: TablesListProps) {
	const [filter, setFilter] = useState("");

	const handleClick = (name: string) => {
		// Toggle selection: if already selected, deselect it
		if (selectedTable === name) {
			onTableSelect("");
		} else {
			onTableSelect(name);
		}
	};

	// Filter tables based on the filter text
	const filteredTables = tables.filter((item) =>
		item.name.toLowerCase().includes(filter.toLowerCase()),
	);

	return (
		<div className="flex flex-col h-full">
			{/* Filter input */}
			<div className="p-3 border-b border-sand-300 dark:border-slate-700">

					<div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none text-slate-500 dark:text-slate-400">
						<SearchSVG />
					</div>
					<input
						type="text"
						value={filter}
						onChange={(e) => setFilter(e.target.value)}
						placeholder="Filter tables..."
						className="w-full pl-9 pr-3 py-1.5 text-sm rounded-lg border border-sand-300 dark:border-slate-600 bg-white dark:bg-slate-900 text-slate-900 dark:text-slate-100 placeholder-slate-500 dark:placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-warm-500 dark:focus:ring-warm-600 focus:border-transparent"
					/>
				</div>
				{filter && (
					<div className="mt-2 text-xs text-slate-600 dark:text-slate-400">
						{filteredTables.length} of {tables.length} table
						{tables.length !== 1 ? "s" : ""}
					</div>
				)}
			</div>

			{/* Tables list */}
			<div className="p-3 space-y-1 overflow-y-auto flex-1">
				{filteredTables.map((item) => {
					const isSelected = selectedTable === item.name;

					return (
						<button
							type="button"
							key={item.name}
							id={item.name}
							onClick={() => handleClick(item.name)}
							className={`
              w-full flex items-center gap-2 px-3 py-1 rounded-lg text-left
              font-medium text-sm transition-all duration-200
              ${
								isSelected
									? "bg-warm-100 dark:bg-warm-900/30 text-warm-800 dark:text-warm-300 shadow-sm"

							</span>

import { identify } from "https://esm.sh/sql-query-identifier@2.7.0";
import type { ParsedStatement } from "./types.ts";

/**
 * Parse SQL code into individual statements with their positions
 */
export function parseSQLStatements(code: string): ParsedStatement[] {
	return identify(code, {
		dialect: "sqlite",
		strict: false,
	});
}

/**
 * Select the SQL statement at a given cursor position
 */
export function selectStatementFromPosition(
	statements: ParsedStatement[],
	position: number,
): ParsedStatement | undefined {
	for (const statement of statements) {
		if (statement.end + 1 >= position) {
			return statement;
		}
	}
	return undefined;
}

/**
 * Get the SQL statement at the current cursor position
 * If no statement found, returns empty string
 */
export function getStatementAtPosition(
	code: string,
	cursorPosition: number,
): string {
	const statements = parseSQLStatements(code);
	const statement = selectStatementFromPosition(statements, cursorPosition);
	return statement?.text || "";
}

/**
 * Generate a default SELECT query for a table or view
 */
export function generateDefaultQuery(
	name: string,
	_type: "table" | "view",
): string {
	return `SELECT * FROM ${name} LIMIT 100;`;
}

# SQLite Explorer (vReact)

_Look ma, no HTMX!_

A modern, web-based SQLite database browser and query tool built for
[Val Town](https://val.town). Execute SQL queries, browse database schema, and
export results with a clean, intuitive interface.

## Features

- 🔍 **Database Schema Browser** - View all tables and views with visual icons
- ✨ **SQL Code Editor** - Syntax highlighting with CodeMirror
- ⚡ **Query Execution** - Execute SQL queries with real-time results
- 📊 **Paginated Results** - Handle large datasets (100 rows per page)
- 💾 **Export Data** - Download results as CSV or JSON
- 🎨 **Dark Mode** - Automatic theme switching with warm color palette
- 💬 **Query Persistence** - Saves queries per table in localStorage
- ⌨️ **Keyboard Shortcuts** - `Cmd/Ctrl+Enter` to run queries
- 🔒 **OAuth Authentication** - Secure Val Town authentication
- 🛡️ **Error Boundaries** - Graceful error handling and recovery

## Tech Stack

### Frontend

- **React 18.2.0** - UI framework
- **CodeMirror 6** - SQL editor with syntax highlighting
- **TailwindCSS** - Styling (via Play CDN)
- **TypeScript** - Type safety with strict mode

### Backend

- **Hono 4** - Lightweight web framework
- **Turso/libSQL** - SQLite database (via Val Town)
- **Deno** - Runtime environment

### Tools

- **Biome** - Linting and formatting
- **TypeScript Strict Mode** - Enhanced type safety

## Usage

### Running Queries

1. **Select a table** from the sidebar to auto-populate the editor with
   `SELECT * FROM [table]`
2. **Write or edit your SQL query** in the editor
3. **Press `Cmd/Ctrl+Enter`** or click the "Run Query" button
4. **View results** in the paginated table below

### Multi-Statement Queries

The editor supports multiple SQL statements. It will execute the statement at
your cursor position when you press `Cmd/Ctrl+Enter`.

### Exporting Data

1. **Execute a query** to get results
2. **Click "Download CSV"** or **"Download JSON"**
3. File downloads automatically with the results

### Keyboard Shortcuts

- `Cmd/Ctrl+Enter` - Execute query at cursor
- Click table name - Auto-populate SELECT query

## API Endpoints

### `GET /api/schema`

Returns all tables and views in the database.

**Response:**

```json
{
  "tables": [{ "name": "users", "type": "table" }],
  "views": [{ "name": "active_users", "type": "view" }]
}
```

### `POST /api/query`

Executes a SQL query.

**Request:**

```json
{ "query": "SELECT * FROM users LIMIT 10" }
```

**Response:**

```json
{
  "success": true,
  "data": {
    "columns": ["id", "name", "email"],
    "rows": [[1, "Alice", "alice@example.com"]]
  }
}
```

### `GET /api/download/csv?query=<sql>`

Downloads query results as CSV.

### `GET /api/download/json?query=<sql>`

Downloads query results as JSON.

## Limitations

- **Admin tool only** - Not designed for public-facing applications
- **Single database** - Works with Val Town's SQLite instance
- **No query history** - Future enhancement planned
- **100 row pagination** - Configurable in `Table.tsx`

	currentPage: number;
	result: QueryResult;
	setCurrentPage: (value: React.SetStateAction<number>) => void;
}

interface PageButtonProps {
	icon: React.ReactNode;
	onClick: () => void;
	disabled: boolean;
}

const PageButton = ({ icon, onClick, disabled }: PageButtonProps) => (
	<button
		type="button"
		onClick={onClick}
		disabled={disabled}
		className="px-2 py-1 text-sm rounded bg-white dark:bg-slate-700 border border-sand-300 dark:border-slate-600 hover:bg-sand-100 dark:hover:bg-slate-600 disabled:opacity-50 disabled:cursor-not-allowed transition-colors"
	>
		{icon}
	</button>
);

export function PaginationBar({
	currentPage,
	result,
	setCurrentPage,
}: PaginationBarProps) {
	return (
		result &&
		result.rows.length > PAGE_SIZE && (
			<div className="border-b border-sand-300 dark:border-slate-700 bg-sand-50 dark:bg-slate-800 px-4 py-2 flex items-center justify-between flex-shrink-0">
				<div className="text-sm text-slate-600 dark:text-slate-400">
					Showing {(currentPage - 1) * PAGE_SIZE + 1} to{" "}
					{Math.min(currentPage * PAGE_SIZE, result.rows.length)} of{" "}
					{result.rows.length} rows
				</div>
				<div className="flex gap-2">
					<PageButton
						onClick={() => setCurrentPage(1)}
						disabled={currentPage === 1}
						icon={<FirstSVG />}
					/>
					<PageButton
						onClick={() => setCurrentPage((p) => Math.max(1, p - 1))}
						disabled={currentPage === 1}
						icon={<PreviousSVG />}
					/>
					<span className="px-2 py-1 text-sm text-slate-700 dark:text-slate-300">
						Page {currentPage} / {Math.ceil(result.rows.length / PAGE_SIZE)}
					</span>
					<PageButton
						onClick={() =>
							setCurrentPage((p) =>
								Math.min(Math.ceil(result.rows.length / PAGE_SIZE), p + 1),
							)
						}
						disabled={currentPage === Math.ceil(result.rows.length / PAGE_SIZE)}
						icon={<NextSVG />}
					/>
					<PageButton
						onClick={() =>
							setCurrentPage(Math.ceil(result.rows.length / PAGE_SIZE))
						}
						disabled={currentPage === Math.ceil(result.rows.length / PAGE_SIZE)}
						icon={<LastSVG />}
					/>
				</div>
			</div>
		)
	);
}

} from "https://esm.town/v/std/oauth/middleware.ts";
import { sqlite } from "https://esm.town/v/std/sqlite";

import { Hono } from "npm:hono@4";
import { sqlToCSV, sqlToJSON } from "../shared/exportHelpers.ts";
import type { QueryResponse, SchemaResponse } from "../shared/types.ts";
import { getErrorMessage } from "../shared/types.ts";

type UserData = Awaited<ReturnType<typeof getOAuthUserData>>;
type Variables = { user: UserData };

const app = new Hono<{ Variables: Variables }>();

app.onError((err, _c) => {
	throw err;
});

app.get("/frontend/*", (c) => serveFile(c.req.path, import.meta.url));
app.get("/shared/*", (c) => serveFile(c.req.path, import.meta.url));

app.use("/api/*", async (c, next) => {
	const user = await getOAuthUserData(c.req.raw);
	if (!user) return c.json({ error: "Unauthorized. Please log in." }, 401);
	c.set("user", user);
	await next();
});

// API Routes

app.get("/api/schema", async (c) => {
	try {
		const data = await sqlite.batch([
			`SELECT name FROM sqlite_schema WHERE type ='table' AND name NOT LIKE 'sqlite_%';`,
			`SELECT name FROM sqlite_schema WHERE type ='view' AND name NOT LIKE 'sqlite_%';`,
		]);

		const tables = data[0].rows.map((row) => ({
			name: row[0] as string,
			type: "table" as const,
		}));

		const views = data[1].rows.map((row) => ({
			name: row[0] as string,
			type: "view" as const,
		}));

		const response: SchemaResponse = { tables, views };
		return c.json(response);
	} catch (err) {
		return c.json({ error: getErrorMessage(err) }, 500);
	}
});

app.post("/api/query", async (c) => {
	try {
		const body = await c.req.json();
		const query = body.query as string;

		if (!query || !query.trim()) {
			return c.json({ success: false, error: "Query is required" }, 400);
		}

		const result = await sqlite.execute(query);

		const response: QueryResponse = {
			success: true,
			data: {
				columns: result.columns,
				rows: result.rows,
			},
		};

		return c.json(response);
	} catch (err) {
		return c.json({
			success: false,
			error: getErrorMessage(err),
		});
	}
});

app.get("/api/download/csv", async (c) => {
	try {
		const query = c.req.query("query");

		if (!query) {
			return c.text("Query parameter is required", 400);
		}

		const result = await sqlite.execute(query);
		const csv = sqlToCSV({ columns: result.columns, rows: result.rows });

		return new Response(csv, {
			headers: {
				"Content-Type": "text/csv",
				"Content-Disposition": 'attachment; filename="results.csv"',
			},
		});
	} catch (err) {
		return c.text(`Failed to download CSV: ${getErrorMessage(err)}`, 500);
	}
});

app.get("/api/download/json", async (c) => {
	try {
		const query = c.req.query("query");
		if (!query) return c.text("Query parameter is required", 400);

		const result = await sqlite.execute(query);
		const jsonData = sqlToJSON({
			columns: result.columns,
			rows: result.rows,
		});

		return Response.json(jsonData, {
			headers: {
				"Content-Disposition": 'attachment; filename="results.json"',
			},
		});
	} catch (err) {
		return c.text(`Failed to download JSON: ${getErrorMessage(err)}`, 500);
	}
});

app.get("/", async (c) => {
	try {
		const html = await readFile("/frontend/index.html", import.meta.url);
		return c.html(html);
	} catch (err) {
		return c.text(`Error loading application: ${getErrorMessage(err)}`, 500);
	}
});

export default oauthMiddleware(app.fetch);

import { Parser } from "https://esm.sh/@json2csv/plainjs";
import type { QueryResult, SQLValue } from "./types.ts";

export const sqlToJSON = (result: QueryResult): Record<string, SQLValue>[] => {
	const { columns, rows } = result;

	return rows.map((row) =>
		columns.reduce(
			(obj, col, index) => {
				obj[col] = row[index];
				return obj;
			},
			{} as Record<string, SQLValue>,
		),
	);
};

export const sqlToCSV = (result: QueryResult): string => {
	const { columns } = result;
	const records = sqlToJSON(result);
	return new Parser({ fields: columns }).parse(records);
};

/** @jsxImportSource https://esm.sh/react@18.2.0 */

import React from "https://esm.sh/react@18.2.0";

interface Props {
	children: React.ReactNode;
	fallback?: React.ReactNode;
}

interface State {
	hasError: boolean;
	error: Error | null;
}

export class ErrorBoundary extends React.Component<Props, State> {
	constructor(props: Props) {
		super(props);
		this.state = { hasError: false, error: null };
	}

	static getDerivedStateFromError(error: Error): State {
		return { hasError: true, error };
	}

	override componentDidCatch(error: Error, errorInfo: React.ErrorInfo): void {
		// In production, you could send to error tracking service
		// For now, errors are just stored in state and displayed
		void error;
		void errorInfo;
	}

	override render() {
		if (this.state.hasError) {
			if (this.props.fallback) {
				return this.props.fallback;
			}

			return (
				<div className="flex items-center justify-center min-h-screen bg-stone-50 dark:bg-stone-900">
					<div className="max-w-md p-8 bg-white dark:bg-stone-800 rounded-lg shadow-lg border border-stone-200 dark:border-stone-700">
						<h2 className="text-2xl font-bold text-red-600 dark:text-red-400 mb-4">
							Something went wrong
						</h2>
						<p className="text-stone-700 dark:text-stone-300 mb-4">
							The application encountered an unexpected error. Please try
							refreshing the page.
						</p>
						{this.state.error && (
							<details className="mt-4 p-4 bg-stone-100 dark:bg-stone-900 rounded text-sm">
								<summary className="cursor-pointer font-semibold text-stone-600 dark:text-stone-400 mb-2">
									Error Details
								</summary>
								<pre className="text-red-600 dark:text-red-400 overflow-auto whitespace-pre-wrap">
									{this.state.error.message}
									{"\n\n"}
									{this.state.error.stack}
								</pre>
							</details>
						)}
						<button
							type="button"
							onClick={() => window.location.reload()}
							className="mt-4 px-4 py-2 bg-amber-600 hover:bg-amber-700 text-white rounded-md transition-colors"
						>
							Refresh Page
						</button>
					</div>
				</div>
			);
		}

		return this.props.children;
	}
}

import { useEffect, useState } from "https://esm.sh/react@18.2.0";
import type { QueryResult, SchemaItem } from "../shared/types.ts";
import { AppHeader } from "./components/AppHeader.tsx";
import { EditorSection } from "./components/EditorSection.tsx";
import { PaginationBar } from "./components/PaginationBar.tsx";
import { Table } from "./components/Table.tsx";
import { TablesList } from "./components/TablesList.tsx";
import { useCodeEditor } from "./hooks/useCodeEditor.ts";
import { useDownload } from "./hooks/useDownload.ts";
import { useQueryExecution } from "./hooks/useQueryExecution.ts";
import { useTableSelection } from "./hooks/useTableSelection.ts";

export function App() {
	const [tables, setTables] = useState<SchemaItem[]>([]);
	const [selectedTable, setSelectedTable] = useState<string | null>(null);
	const [isLoadingSchema, setIsLoadingSchema] = useState(true);
	const [isAuthenticated, setIsAuthenticated] = useState(true);
	const [isSidebarOpen, setIsSidebarOpen] = useState(true);
	const [currentPage, setCurrentPage] = useState(1);

	const { code, setCode, getActiveStatement, editorRef } = useCodeEditor();
	const { result, error, isLoading, executeQuery } = useQueryExecution();
	const { saveQuery } = useTableSelection(selectedTable, setCode);
	const { downloadQuery } = useDownload();

	useEffect(() => {
		setCurrentPage(1);
	}, [result]);

	useEffect(() => {
		const loadSchema = async () => {
			try {
				const response = await fetch("/api/schema");
				if (response.status === 401) {
					setIsAuthenticated(false);
					return;
				}
				const data = await response.json();
				const allItems = [...data.tables, ...data.views];
				setTables(allItems);
				setIsAuthenticated(true);
			} catch (_err) {
				setTables([]);
			} finally {
				setIsLoadingSchema(false);
			}
		};

		loadSchema();
	}, []);

	useEffect(() => {
		const handleResize = () => {
			if (window.innerWidth < 1024) {
				setIsSidebarOpen(false);
			} else {
				setIsSidebarOpen(true);
			}
		};
		handleResize();
		window.addEventListener("resize", handleResize);
		return () => window.removeEventListener("resize", handleResize);
	}, []);

	const handleRun = () => {
		const statement = getActiveStatement();
		if (statement) {
			executeQuery(statement);
			if (selectedTable) {
				saveQuery(selectedTable, statement);
			}
		}
	};

	const handleDownloadJSON = async () => {
		const statement = getActiveStatement();
		if (statement) {
			await downloadQuery(statement, "json");
		}
	};

	const handleDownloadCSV = async () => {
		const statement = getActiveStatement();
		if (statement) {
			await downloadQuery(statement, "csv");
		}
	};

	if (!isAuthenticated) {
		return (
			<div className="h-screen flex flex-col overflow-hidden">

							code={code}
							onCodeChange={setCode}
							onRun={handleRun}
							onDownloadJSON={handleDownloadJSON}
							onDownloadCSV={handleDownloadCSV}
							isLoading={isLoading}
							editorRef={editorRef}
						/>
					</div>

					<PaginationBar
						currentPage={currentPage}
						result={result ?? ({ columns: [], rows: [] } as QueryResult)}
						setCurrentPage={setCurrentPage}
					/>

					<div className="flex-1 overflow-auto bg-white dark:bg-slate-900 min-h-0">
						{error && (
							<div className="text-red-600 dark:text-red-400 font-semibold bg-red-50 dark:bg-red-900/20 rounded-lg border border-red-200 dark:border-red-800">
								{error}
							</div>
						)}
						{result && (
							<Table
								headers={result.columns}
								rows={result.rows}
								currentPage={currentPage}
							/>
						)}
						{!error && !result && (
							<div className="flex items-center justify-center h-full text-sand-500 dark:text-slate-400">
								Run a query to see results here
							</div>
						)}
					</div>
				</div>
			</main>
		</div>
	);
}

## Val Town Utility Functions

Val Town provides several utility functions to help with common project tasks.

### Importing Utilities

Always import utilities with version pins to avoid breaking changes:

```ts
import {
```

### Available Utilities

#### **serveFile** - Serve project files with proper content types

For example, in Hono:

```ts
// serve all files in frontend/ and shared/
app.get("/frontend/*", (c) => serveFile(c.req.path, import.meta.url));
```

#### **listFiles** - List all files in the project

```ts
const files = await listFiles(import.meta.url);
```

#### **parseProject** - Extract information about the current project from import.meta.url

This is useful for including info for linking back to a val, ie in "view source"

```ts
const projectVal = parseProject(import.meta.url);
```

data on the client, run it in the server and pass it to the client by splicing
it into the HTML page or by making an API request for it.

## Val Town Platform Specifics

- **Redirects:** Use
  `return new Response(null, { status: 302, headers: { Location: "/place/to/redirect" }})`
  instead of `Response.redirect` which is broken
- **Images:** Avoid external images or base64 images. Use emojis, unicode
  symbols, or icon fonts/libraries instead
- **AI Image:** To inline generate an AI image use:
  `<img src="https://maxm-imggenurl.web.val.run/the-description-of-your-image" />`
- **Storage:** DO NOT use the Deno KV module for storage
- **Browser APIs:** DO NOT use the `alert()`, `prompt()`, or `confirm()` methods
- **Weather Data:** Use open-meteo for weather data (doesn't require API keys)
  unless otherwise specified
- **View Source:** Add a view source link by importing & using
  `import.meta.url.replace("ems.sh", "val.town)"` (or passing this data to the
  client) and include `target="_top"` attribute
- **Error Debugging:** Add
  `<script src="https://esm.town/v/std/catch"></script>` to HTML to capture
  client-side errors
- **Error Handling:** Only use try...catch when there's a clear local
  resolution; Avoid catches that merely log or return 500s. Let errors bubble up
  with full context
- **Environment Variables:** Use `Deno.env.get('keyname')` when you need to, but
  generally prefer APIs that don't require keys
- **Imports:** Use `https://esm.sh` for npm and Deno dependencies to ensure
  compatibility on server and browser
- **Storage Strategy:** Only use backend storage if explicitly required; prefer
  simple static client-side sites
- **React Configuration:** When using React libraries, pin versions with
  `?deps=react@18.2.0,react-dom@18.2.0` and start the file with
  `/** @jsxImportSource https://esm.sh/react@18.2.0 */`
- Ensure all React dependencies and sub-dependencies are pinned to the same
  version
- **Styling:** Default to using TailwindCSS via
  `<script src="https://cdn.twind.style" crossorigin></script>` unless otherwise
  specified

## Common Gotchas and Solutions

1. **Environment Limitations:**
   - Val Town runs on Deno in a serverless context, not Node.js
   - Code in `shared/` must work in both frontend and backend environments
   - Cannot use `Deno` keyword in shared code
   - Use `https://esm.sh` for imports that work in both environments

2. **SQLite Peculiarities:**
   - Limited support for ALTER TABLE operations
   - Create new tables with updated schemas and copy data when needed
   - Always run table creation before querying

3. **React Configuration:**
   - All React dependencies must be pinned to 18.2.0
   - Always include `@jsxImportSource https://esm.sh/react@18.2.0` at the top of
     React files
   - Rendering issues often come from mismatched React versions

4. **File Handling:**
   - Val Town only supports text files, not binary
   - Use the provided utilities to read files across branches and forks
   - For files in the project, use `readFile` helpers

5. **API Design:**
   - `fetch` handler is the entry point for HTTP vals
   - Run the Hono app with
     `export default app.fetch // This is the entry point for HTTP vals`
